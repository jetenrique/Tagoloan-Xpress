+# Tagoloan Xpress rider UI update
+
+Prepared against GitHub commit `b8de1bd` on 10 November 2026. This is a local rider prototype, not a backend implementation. Existing customer screens and Gradle versions are preserved.
+
+## Included flows
+
+- Login → dashboard; registration → pending review → explicit demo approval → login → profile photo setup.
+- Available request → details → acceptance → store arrival → receipt and actual cost → delivery → customer arrival → handover proof and confirmations → completion.
+- History filters, trip receipts, receipt/handover image previews, profile photo, vehicle and document editing.
+- Local chat, quick rider updates, notification previews, and user-controlled native SMS composer/dialer.
+- One state model for delivery status, confirmed costs, completed trips and earnings. Records survive app restarts and sign-out.
+
+## Applying the download with Copilot
+
+Extract `Tagoloan-Rider-Update.zip` into a temporary folder. Copy that extracted folder into your Android Studio project, then paste:
+
+```text
+Read Tagoloan-Rider-Update/APPLY_WITH_COPILOT.md and apply the supplied
+rider-ui-update.patch to this project. The implementation is already written;
+do not regenerate the screens. Check that the patch applies before changing
+files. Preserve any newer local work and stop to report conflicts if needed.
+Then run :app:assembleDebug and :app:testDebugUnitTest. With the emulator
+connected, run the two RiderFlowTest instrumentation tests. Report actual
+results separately for compilation, unit tests, and emulator tests.
+```
+
+Do not replace the entire `app` folder with a partial folder. The patch contains all source, resource, manifest, and test changes.
+
+## Verification on your Mac
+
+From the project root:
+
+```sh
+bash gradlew :app:assembleDebug :app:testDebugUnitTest
+bash gradlew :app:connectedDebugAndroidTest -Pandroid.testInstrumentationRunnerArguments.class=com.tagoloanxpress.rider.RiderFlowTest
+```
+
+The instrumentation tests use a separate preferences file and do not reset your normal demo records.
+
+Manual checks:
+
+1. Enter any nonempty rider login. No actual credentials are verified.
+2. Accept the sample request. Verify BUSY status and removal from Available Requests.
+3. Mark store arrival. Enter `195.00`, select/capture a receipt image, and check `₱255.00` total.
+4. Cancel a photo picker and verify the previous selection remains. Back/reopen must retain the draft.
+5. Start delivery. Receipt viewing must now be read-only; the confirmed product cost must not change.
+6. Mark customer arrival. Completion must remain disabled until a readable handover photo and both receipt/payment confirmations are present.
+7. Complete once. History must contain one trip with `₱195.00` product cost, `₱60.00` earnings, and `₱255.00` cash collected. No duplicate fee should be added.
+8. Restart the app, log in again and verify that history remains. Cancel a sign-out dialog during a delivery and verify no draft is cleared.
+9. Check Today/This Week filters, trip photos, profile/vehicle edits, notification actions, local chat and keyboard scrolling.
+10. Test registration separately: all three document photos and valid fields are required. Pending applicants cannot access requests. Only the labeled demo approval control advances the preview.
+
+The app supports one local demo rider profile. Submitting a new application replaces that local demo profile. Finish testing the delivery/history flow before testing registration.
+
+## Design and runtime limits
+
+Screens follow the supplied Figma references. History/trip-log Figma frames are flattened reference images, so their content is rebuilt as native interactive UI. Android device sizes and fonts still require visual comparison on the emulator; pixel-perfect fidelity is not asserted.
+
+The map remains a route placeholder. No live GPS, authentication, password changes, admin approvals, document uploads, payment transfers, FCM delivery, encrypted messaging, or automatic connection failover is implemented. Chat messages are saved locally. SMS/call controls open another app and require a recipient number chosen by you; they do not send or place calls automatically.
+
+Real admin approval must replace the demo approval control before production. Photo attachments are app-private local copies. Clear app data/uninstall removes these local records.
+

+) {

+    val available: Boolean get() = approval == Approval.APPROVED && profileReady && accepting && stage == DeliveryStage.AVAILABLE
+    val total: Long get() = confirmedCost + FEE
+    val earnings: Long get() = trips.sumOf { it.fee }
+    val canComplete: Boolean get() = stage == DeliveryStage.AT_CUSTOMER && confirmedCost > 0 && receipt.isNotBlank() && handover.isNotBlank() && receiptHandedOver && cashCollected
+
+    fun accept(): RiderState = if (available) copy(stage = DeliveryStage.TO_STORE) else this
+    fun arriveAtStore(): RiderState = if (stage == DeliveryStage.TO_STORE) copy(stage = DeliveryStage.BUYING) else this
+    fun startDelivery(cost: Long): RiderState = if (stage == DeliveryStage.BUYING && cost in 1..99999999L && receipt.isNotBlank()) copy(confirmedCost = cost, stage = DeliveryStage.DELIVERING) else this
+    fun arriveAtCustomer(): RiderState = if (stage == DeliveryStage.DELIVERING) copy(stage = DeliveryStage.AT_CUSTOMER) else this
+    fun complete(now: Long): RiderState {
+        if (!canComplete || trips.any { it.id == REQUEST_ID }) return this
+        return copy(
+            stage = DeliveryStage.COMPLETED, accepting = true,
+            trips = listOf(CompletedTrip(REQUEST_ID, confirmedCost, FEE, receipt, handover, now)) + trips,
+        )
+    }

+        const val REQUEST_ID = "TGX-8921"
+    }

+fun parseRiderCentavos(input: String): Long? {
+    val value = input.trim()
+    if (!Regex("""\d{1,6}(\.\d{1,2})?""").matches(value)) return null
+    return try { BigDecimal(value).movePointRight(2).longValueExact().takeIf { it in 1..99999999L } }
+    catch (_: ArithmeticException) { null } catch (_: NumberFormatException) { null }

+class RiderStore(context: Context, preferencesName: String = "rider_ui_demo_v1") {
+    private val prefs = context.applicationContext.getSharedPreferences(preferencesName, Context.MODE_PRIVATE)
+    var state by mutableStateOf(read())
+        private set
+
+    fun update(transform: (RiderState) -> RiderState) {
+        val next = transform(state)
+        if (next == state) return
+        state = next
+        val s = JSONObject().apply {
+            put("name", next.name); put("mobile", next.mobile); put("username", next.username)
+            put("plate", next.plate); put("vehicle", next.vehicle); put("photo", next.photo)
+            put("documents", JSONArray(next.documents)); put("approval", next.approval.name)
+            put("profileReady", next.profileReady); put("accepting", next.accepting)
+            put("stage", next.stage.name); put("draftCost", next.draftCost); put("receipt", next.receipt)
+            put("confirmedCost", next.confirmedCost); put("handover", next.handover)
+            put("receiptHandedOver", next.receiptHandedOver); put("cashCollected", next.cashCollected)
+            put("messages", JSONArray(next.messages)); put("notificationsRead", next.notificationsRead)
+            put("trips", JSONArray().apply { next.trips.forEach { t -> put(JSONObject().apply {
+                put("id", t.id); put("productCost", t.productCost); put("fee", t.fee)
+                put("receipt", t.receipt); put("handover", t.handover); put("completedAt", t.completedAt)
+            }) } })
+        }
+        prefs.edit().putString("state", s.toString()).apply()
+    }
+
+    private fun read(): RiderState {
+        val raw = prefs.getString("state", null) ?: return RiderState()
+        return try {
+            val j = JSONObject(raw)
+            fun strings(key: String): List<String> = j.optJSONArray(key)?.let { a -> (0 until a.length()).map { a.getString(it) } } ?: emptyList()
+            RiderState(
+                name = j.getString("name"), mobile = j.getString("mobile"), username = j.getString("username"),
+                plate = j.getString("plate"), vehicle = j.getString("vehicle"), photo = j.optString("photo"),
+                documents = strings("documents").let { if (it.size == 3) it else listOf("", "", "") },
+                approval = Approval.valueOf(j.getString("approval")), profileReady = j.getBoolean("profileReady"),
+                accepting = j.getBoolean("accepting"), stage = DeliveryStage.valueOf(j.getString("stage")),

+package com.tagoloanxpress.rider
+
+import com.tagoloanxpress.rider.model.*
+import org.junit.Assert.*
+import org.junit.Test
+
+class RiderStateTest {
+    @Test fun pendingAndOfflineRidersCannotAccept() {
+        val pending = RiderState(approval = Approval.PENDING)
+        assertEquals(pending, pending.accept())
+        val offline = RiderState(accepting = false)
+        assertEquals(offline, offline.accept())
+        val missingProfile = RiderState(profileReady = false)
+        assertEquals(missingProfile, missingProfile.accept())
+    }
+
+    @Test fun deliveryStagesCannotBeSkipped() {
+        val available = RiderState()
+        assertEquals(available, available.arriveAtStore())
+        assertEquals(available, available.startDelivery(19500))
+        assertEquals(available, available.arriveAtCustomer())
+        assertEquals(available, available.complete(100))
+        val toStore = available.accept()
+        assertTrue(toStore.busy)
+        assertFalse(toStore.available)
+        assertEquals(toStore, toStore.startDelivery(19500))
+    }
+
+    @Test fun receiptAndPositiveAmountAreRequired() {
+        val buying = RiderState().accept().arriveAtStore()
+        assertEquals(buying, buying.startDelivery(19500))
+        val receipt = buying.copy(receipt = "file:///receipt.jpg")
+        assertEquals(receipt, receipt.startDelivery(0))
+        assertEquals(receipt, receipt.startDelivery(-1))
+        assertEquals(receipt, receipt.startDelivery(100000000))
+        assertEquals(DeliveryStage.DELIVERING, receipt.startDelivery(19500).stage)
+    }
+
+    @Test fun completionRequiresPhotoAndBothConfirmations() {
+        val arrived = atCustomer()
+        assertFalse(arrived.canComplete)
+        assertEquals(arrived, arrived.complete(100))
+        assertFalse(arrived.copy(handover = "photo").canComplete)
+        assertFalse(arrived.copy(handover = "photo", receiptHandedOver = true).canComplete)
+        assertTrue(arrived.copy(handover = "photo", receiptHandedOver = true, cashCollected = true).canComplete)
+    }
+
+    @Test fun completionRecordsFeeNotCashCollectionAndIsIdempotent() {
+        val ready = atCustomer().copy(handover = "file:///proof.jpg", receiptHandedOver = true, cashCollected = true)
+        val done = ready.complete(12345)
+        assertEquals(DeliveryStage.COMPLETED, done.stage)
+        assertFalse(done.busy)
+        assertTrue(done.accepting)
+        assertFalse(done.available)
+        assertEquals(6000L, done.earnings)
+        assertEquals(25500L, done.trips.single().total)
+        assertEquals(19500L, done.trips.single().productCost)
+        assertEquals("file:///proof.jpg", done.trips.single().handover)
+        assertEquals(12345L, done.trips.single().completedAt)
+        assertEquals(done, done.complete(99999))
+        assertEquals(done, done.accept())
+    }
+
+    @Test fun confirmedCostCannotBeRestartedDuringDelivery() {
+        val delivering = RiderState(stage = DeliveryStage.BUYING, receipt = "photo").startDelivery(19500)
+        assertEquals(delivering, delivering.startDelivery(34000))
+        assertEquals(19500L, delivering.confirmedCost)
+    }
+
+    @Test fun declinedRequestIsUnavailable() {
+        val declined = RiderState(stage = DeliveryStage.DECLINED)
+        assertFalse(declined.available)
+        assertEquals(declined, declined.accept())
+    }
+
+    @Test fun moneyParsingRejectsRoundingAndInvalidValues() {
+        assertEquals(19500L, parseRiderCentavos("195.00"))
+        assertEquals(1L, parseRiderCentavos("0.01"))
+        assertEquals(99999999L, parseRiderCentavos("999999.99"))
+        listOf("", "0", "-1", "1.999", "NaN", "1e3", "1,000", "1000000", ".50").forEach { assertNull(it, parseRiderCentavos(it)) }
+        assertEquals("₱255.00", riderMoney(25500))
+    }
+
+    private fun atCustomer(): RiderState = RiderState().accept().arriveAtStore().copy(receipt = "file:///receipt.jpg").startDelivery(19500).arriveAtCustomer()
+}
